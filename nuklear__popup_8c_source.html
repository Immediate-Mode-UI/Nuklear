11 struct nk_rect anchor, enum nk_popup_fit fit, float known_h,
12 nk_bool stay_up)
13{
14 struct nk_rect d;
15 float overlap;
16 float space_below;
17 float space_above;

19 float d_bottom;

21 float need_h;

23 if (!ctx) return body;
24 d = ctx->display_bounds;
25 if (d.w <= 0 || d.h <= 0) return body;
26
27 d_right = d.x + d.w;
28 d_bottom = d.y + d.h;
29 req_h = body.h;
30 /* last_h is unclamped layout height (every row, including those that
31 * would scroll). Combos pass a smaller panel (e.g. 200 with 370px of
32 * items); menus pass a larger max (e.g. 600 with 80px of trees).
33 * Judge the visible panel: min(content, caller max). */
34 if (known_h > req_h)
35 known_h = req_h;
36 need_h = (known_h > 0.0f) ? known_h : req_h;
37 overlap = (anchor.y + anchor.h) - body.y;
38
39 /* horizontal: flip/align, then slide. Do not shrink width; dynamic
40 * rows (e.g. the overview color contextual) size to the panel and
41 * would smush. If the surface is narrower than the popup, clip. */
42 if (fit == NK_POPUP_FIT_FLIP) {
43 if (body.x + body.w > d_right)
44 body.x = anchor.x + anchor.w - body.w;
45 } else if (fit == NK_POPUP_FIT_TOOLTIP) {
46 if (body.x + body.w > d_right) {
47 float flipped = anchor.x - body.w;
48 if (flipped >= d.x)
49 body.x = flipped;
50 }
51 }
52 if (body.x + body.w > d_right)
53 body.x = d_right - body.w;
54 if (body.x < d.x)
55 body.x = d.x;
56
57 space_below = d_bottom - body.y;
58 space_above = anchor.y - d.y;
59 if (space_below < 0.0f) space_below = 0.0f;
60 if (space_above < 0.0f) space_above = 0.0f;
61
62 if (fit == NK_POPUP_FIT_FLIP) {
63 int flip = 0;
64 /* Caller size.y is a maximum (ADVANCED is 600px for expanded trees),
65 * not the current content. Flip only when last frame's actual height
66 * does not fit below. First open (known_h == 0) always drops down.
67 * stay_up keeps a drop-up for the rest of this open so collapsing
68 * a tree does not jump back to a dropdown. */
69 if (stay_up && space_above > 0.0f)
70 flip = 1;
71 else if (known_h > 0.0f && known_h > space_below) {
72 if (known_h <= space_above)
73 flip = 1;
74 else if (space_above > space_below)
75 flip = 1;
76 }
77 if (flip) {
78 float use_h = req_h;
79 if (use_h > space_above) use_h = space_above;
80 if (known_h < use_h) use_h = known_h;
81 if (use_h < 1.0f) use_h = 1.0f;
82 body.y = anchor.y - use_h;
83 if (overlap > 0.0f) body.y += overlap;
84 body.h = use_h;
85 } else if (known_h > space_below) {
86 body.h = (space_below > 1.0f) ? space_below : 1.0f;
87 }
88 } else if (fit == NK_POPUP_FIT_SLIDE) {
89 /* size.y is a maximum, not current content. First open (known_h == 0)
90 * stays on the click; later frames slide only if actual height
91 * does not fit below. Sliding with the max left the DYNAMIC-shrunk
92 * panel floating above the cursor. */
93 if (known_h > 0.0f && known_h > space_below) {
94 body.y = d_bottom - known_h;
95 if (body.y < d.y)
96 body.y = d.y;
97 if (body.y + known_h > d_bottom)
98 body.h = d_bottom - body.y;
99 else body.h = known_h;
100 if (body.h < 1.0f) body.h = 1.0f;
101 }
102 } else {
103 /* tooltip: flip then slide; do not shrink (clipping is acceptable) */
104 if (body.y + need_h > d_bottom) {
105 float flipped = anchor.y - need_h;
106 if (flipped >= d.y)
107 body.y = flipped;
108 }
109 if (body.y + need_h > d_bottom)
110 body.y = d_bottom - need_h;
111 if (body.y < d.y)
112 body.y = d.y;
113 }
114 return body;
115}
